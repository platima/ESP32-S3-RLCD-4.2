# ESP32-S3 RLCD clock: settings file
#
# The clock found no settings file on this card, so it wrote this one.  Every setting
# below is commented out (it starts with #) and shows the value the clock uses now.
# To change one: remove the # in front of its line, edit the value, save the file,
# put the card back in the clock and restart it.  The clock saves what it reads in its
# own flash, so the card can be taken out again afterwards.
#
# The rules
#   * One "name = value" per line.  Spaces around the = do not matter.
#   * Values can be in quotes or not: "My Network" and My Network mean the same.
#     Use quotes if a value contains a # or begins or ends with a space.
#   * Names ignore case, spaces, - and _: wifi_ssid, WiFi SSID and Wi-Fi-SSID are one setting.
#   * Lines starting with # or ; are comments.
#   * A setting you leave out keeps whatever the clock has now.
#   * A line with nothing after the = (for example "timezone =") makes the clock forget
#     that saved setting and go back to its built-in default.
#   * reset_all = yes forgets every saved setting first; remove that line afterwards.
#   * What the clock does not understand is ignored and listed on its Power and settings
#     page (the second info page).
#
# WiFi passwords are plain text in this file and in the clock's flash: keep the card
# to yourself.

# ---- WiFi --------------------------------------------------------
#
# on or off.  Off keeps the radio off: no network time, weather or Spotify, and less
# power drawn.  Without network time the clock runs on its own crystal and drifts
# (see the note at the end of this file).
# wifi = on
#
# Name of your WiFi network (2.4 GHz only).
# wifi_ssid = ""
#
# Its password.  Kept as plain text here and in the clock's flash.  For an open
# network write two quotes: wifi_password = ""
# wifi_password = "your password"
#
# A second network, for when the first cannot be joined (a phone's hotspot, another
# router).  The clock tries the main network first.  While it is on the backup it looks
# for the main one every 10 minutes and goes back to it.  Empty = no backup.
# wifi_backup_ssid = ""
#
# Its password (for an open network write two quotes).
# wifi_backup_password = "your password"
#
# The clock's name on your network: letters, digits and '-'.
# hostname = "rlcd-clock"
#
# Time server to try first (pool.ntp.org, time.cloudflare.com and time.google.com
# follow).  Empty = the built-in list.
# ntp_server = ""
#
# normal or max.  Max lets the radio sleep longer between messages: a little less
# power, but every reply can come up to a third of a second late, the network time
# included (look at "last step" on the Info page).  The Spotify setup page always
# runs in normal.
# wifi_power_save = normal

# ---- Units and formats -------------------------------------------
#
# metric (degrees C, km/h) or imperial (degrees F, mph).
# units = metric
#
# 24h or 12h (12h shows AM or PM).
# time_format = 24h
#
# iso (2026-10-04), dmy (04/10/2026), mdy (10/04/2026), dmy-dot (04.10.2026),
# d-mon-y (4 Oct 2026) or mon-d-y (Oct 4, 2026).
# date_format = iso
#
# on or off.  Shows the ISO week number and the day of the year after the weekday.
# show_week = on

# ---- Location ----------------------------------------------------
#
# Exact coordinates of the clock, in degrees: -31.952 and 115.861, or 31.952 S and
# 115.861 E.  They win over the place name below.
# latitude = 0
# longitude = 0
#
# What the status bar shows.  Empty = the name that was found.
# location_label = ""
#
# A place to look up (Open-Meteo), such as Perth or "Paris, France".  Used when no
# coordinates are set.
# location = ""
#
# Overrides the automatic time zone: an IANA name (Australia/Perth) or a POSIX rule
# (AWST-8).  Empty = follow the location, which needs WiFi: with wifi = off set it here.
# timezone = ""

# ---- Spotify -----------------------------------------------------
#
# on or off.  Spotify also needs the Client ID below.
# spotify = on
#
# The Client ID of your app in the Spotify developer dashboard (see the README).
# spotify_client_id = "your client id"

# ---- Sensors and battery -----------------------------------------
#
# Degrees C added to the indoor temperature to make up for the board warming the
# sensor (the factory value is -4.0).
# indoor_offset = -4
#
# auto or none.  Say none when no battery is fitted (running from USB): the gauge
# then shows USB, nothing is estimated and the clock never shuts down for a low
# battery.  The clock cannot tell by itself: on USB the empty battery socket reads
# like a full battery.
# battery = auto
#
# Capacity of your battery in mAh, so the Power and settings page can show the
# average current.  0 = unknown.
# battery_capacity_mah = 0
#
# on or off.  Shut down (deep sleep, with a message on the screen) before the battery
# is flat, so a LiPo is not ruined.  It wakes by itself once charging has brought the
# battery back, or when you press a button.
# low_battery_shutdown = on
#
# Battery voltage, as measured while the clock runs, at which it shuts down:
# 3.10 to 3.60 (default 3.30).
# battery_cutoff_v = 3.3

# ---- Power -------------------------------------------------------
#
# 80, 160 or 240.  80 uses the least power and is plenty for a clock.
# cpu_mhz = 80

# ---- Weather -----------------------------------------------------
#
# Minutes between weather updates, 5 to 240 (default 15).
# weather_interval_min = 15

# ---- Clock drift when WiFi is off ---------------------------------
#
# With wifi = off the clock cannot ask the network for the time.  It starts from the
# time kept by its real-time clock chip and then counts with its own crystal, which
# is good for a second or two a day (10 ppm is about 0.9 seconds a day) and depends on
# temperature.  Over weeks that adds up to a visible error; nothing corrects it until
# WiFi has been on for a few minutes again.  If you leave WiFi off, switch it on now
# and then to let the clock set itself.
